# Development and CI dependencies.
#
# The model itself needs only numpy. Everything else here exists to test and check
# the repository. Versions are floors rather than pins: this is a research repository
# whose results are pinned by the model hash and the caches, not by the toolchain.
numpy>=1.26
pytest>=8.0
pytest-cov>=5.0
coverage>=7.4
# Lint. Floors rather than pins, as above: the ruleset is narrow enough that a
# newer ruff will not start failing the build on style.
ruff>=0.6
# Shell lint, so tools/run_ci_locally.sh can check the shell scripts the way CI
# does. The wheel bundles the shellcheck binary, so no system package is needed.
shellcheck-py>=0.10
# PDF inspection for tools/check_pdfs.py. Pure Python on purpose: the first version
# of that checker shelled out to poppler, passed here and failed on the runner,
# which had no pdfinfo. A checker with an undeclared system dependency is a checker
# that does not run.
pypdf>=4.0
